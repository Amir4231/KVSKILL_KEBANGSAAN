C2 5.25 Database — 4 Tables + Secret Status Field

MILITARY SECRET: PDF table for Reports omits status but C6/C7 require it. You MUST add status tinyint default 0 (0=public, 1=private). Without it you lose ~10 marks. Also Reports.Category_id + Users.Category_id are nullable FK → categories.

categories — id | category_name

  • id PK bigint
  • category_name varchar(255)
  • Seed: Web Technology, Patisserie and Confectionery, … + categoris.csv

users — id | username UQ | password hash | role | category_id FK nullable

  • id PK
  • username unique, login
  • password hashed (Hash::make)
  • role 0=user 1=admin
  • category_id → categories, nullOnDelete
  • Pre-insert: 1 Admin/1, 2 Alice/0, 3 Bob/0, password=password

reports — id | user_id cascade | category_id | title | description | date | status

  • user_id → users cascade
  • category_id nullable → categories
  • title varchar 255 required
  • description varchar 255 required
  • date date required
  • status 0 public 1 private — REQUIRED

photos — id | report_id cascade | path

  • report_id → reports cascade
  • path varchar 255 e.g. uploads/xxx.jpg
  • Cascade = delete user/report auto-deletes children — saves 30 min

<?php
namespace App\Http\Controllers;
use Illuminate\Http\Request, App\Models\User, App\Models\Category, Illuminate\Support\Facades\Hash;
class AuthController extends Controller {
 public function loginForm(){ return view('login'); }
 public function login(Request $r){
  $r->validate(['username'=>'required','password'=>'required']);
  $u = User::where('username',$r->username)->first();
  if(!$u || !Hash::check($r->password,$u->password))
   return back()->with('error','Username and password not matched.');
  session(['user_id'=>$u->id,'username'=>$u->username,'role'=>$u->role]);
  return $u->role==1 ? redirect('/users') : redirect('/reports');
 }
 public function logout(){ session()->flush(); return redirect('/'); }
 public function registerForm(){ return view('register',['cats'=>Category::all()]); }
 public function register(Request $r){
  $r->validate([
   'username'=>'required|unique:users,username',
   'password'=>'required','confirm'=>'required|same:password',
   'category_id'=>'required|exists:categories,id'],
   ['username.unique'=>'Username already exists.',
    'confirm.same'=>'Password and confirm password must matched.']);
  User::create(['username'=>$r->username,'password'=>Hash::make($r->password),
   'role'=>0,'category_id'=>$r->category_id]);
  return redirect('/')->with('ok','Registered. Please login.');
 }
}

private function resize($src,$dst,$maxW,$maxH){
 [$w,$h,$type]=getimagesize($src); $ratio=min($maxW/$w,$maxH/$h,1);
 $nw=(int)($w*$ratio); $nh=(int)($h*$ratio);
 switch($type){case IMAGETYPE_JPEG:$s=imagecreatefromjpeg($src);break;
  case IMAGETYPE_PNG:$s=imagecreatefrompng($src);break;
  case IMAGETYPE_GIF:$s=imagecreatefromgif($src);break;
  case IMAGETYPE_WEBP:$s=imagecreatefromwebp($src);break;
  default:copy($src,$dst);return;}
 $d=imagecreatetruecolor($nw,$nh);
 if($type==IMAGETYPE_PNG||$type==IMAGETYPE_WEBP){imagealphablending($d,false);imagesavealpha($d,true);}
 imagecopyresampled($d,$s,0,0,0,0,$nw,$nh,$w,$h);
 switch($type){case IMAGETYPE_JPEG:imagejpeg($d,$dst,85);break;
  case IMAGETYPE_PNG:imagepng($d,$dst);break;
  case IMAGETYPE_GIF:imagegif($d,$dst);break;
  case IMAGETYPE_WEBP:imagewebp($d,$dst,85);break;}
 imagedestroy($s);imagedestroy($d);
}
