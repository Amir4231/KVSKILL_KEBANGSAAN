// 1. categories up()/down():
public function up(): void {
  Schema::create('categories', function (Blueprint $table) {
    $table->id();
    $table->string('category_name', 255);
    $table->timestamps();
  });
}
public function down(): void {
  Schema::dropIfExists('categories');
}

// 2. users up()/down():
public function up(): void {
  Schema::create('users', function (Blueprint $table) {
    $table->id();
    $table->string('username', 255)->unique();
    $table->string('password', 255);
    $table->tinyInteger('role')->default(0); // 0=user 1=admin
    $table->foreignId('category_id')->nullable()->constrained('categories')->nullOnDelete();
    $table->timestamps();
  });
}
public function down(): void {
  Schema::dropIfExists('users');
}

// 3. reports up()/down():
public function up(): void {
  Schema::create('reports', function (Blueprint $table) {
    $table->id();
    $table->foreignId('user_id')->constrained('users')->onDelete('cascade');
    $table->foreignId('category_id')->nullable()->constrained('categories')->nullOnDelete();
    $table->string('title', 255);
    $table->string('description', 255);
    $table->date('date');
    $table->tinyInteger('status')->default(0); // 0=public 1=private
    $table->timestamps();
  });
}
public function down(): void {
  Schema::dropIfExists('reports');
}

// 4. photos up()/down():
public function up(): void {
  Schema::create('photos', function (Blueprint $table) {
    $table->id();
    $table->foreignId('report_id')->constrained('reports')->onDelete('cascade');
    $table->string('path', 255);
    $table->timestamps();
  });
}
public function down(): void {
  Schema::dropIfExists('photos');
}

5.2 Auth (C3)

Login validates required; Hash::check; mismatch returns Username and password not matched. Register validates username required|unique, confirm required|same:password (Password and confirm password must matched.), category_id required|exists; duplicate returns Username already exists. Success → / with Registered. Please login. Login view must show both session error and $errors (empty-field validate fails).

AuthController.php — FULL FILE
<?php
namespace App\Http\Controllers;
use Illuminate\Http\Request;
use App\Models\User;
use App\Models\Category;
use Illuminate\Support\Facades\Hash;

class AuthController extends Controller {
  public function loginForm(){ return view('login'); }

  public function login(Request $r){
    $r->validate(['username'=>'required','password'=>'required']);
    $u = User::where('username',$r->username)->first();
    if(!$u || !Hash::check($r->password,$u->password)){
      return back()->with('error','Username and password not matched.');
    }
    session(['user_id'=>$u->id,'username'=>$u->username,'role'=>$u->role]);
    if($u->role==1) return redirect('/users');
    return redirect('/reports');
  }

  public function logout(){
    session()->flush();
    return redirect('/');
  }

  public function registerForm(){
    $cats = Category::all();
    return view('register',compact('cats'));
  }

  public function register(Request $r){
    $r->validate([
      'username'=>'required|unique:users,username',
      'password'=>'required',
      'confirm'=>'required|same:password',
      'category_id'=>'required|exists:categories,id',
    ],['username.unique'=>'Username already exists.','confirm.same'=>'Password and confirm password must matched.']);
    User::create([
      'username'=>$r->username,
      'password'=>Hash::make($r->password),
      'role'=>0,
      'category_id'=>$r->category_id,
    ]);
    return redirect('/')->with('ok','Registered. Please login.');
  }
}

<?php
namespace App\Http\Controllers;
use Illuminate\Http\Request;
use App\Models\Report;
use App\Models\Category;
use App\Models\User;

class ReportController extends Controller {
  public function index(){
    if(!session('user_id')) return redirect('/');
    $reports = Report::where('status',0)->with(['user','photos'])->get()->filter(fn($r)=>$r->photos->count()>0);
    return view('onepage',compact('reports'));
  }

  public function show($id){
    if(!session('user_id')) return redirect('/');
    $rep = Report::with(['user','photos'])->findOrFail($id);
    if($rep->photos->count()==0){
      return redirect('/reports');
    }
    $catName = $rep->user && $rep->user->category_id ? Category::find($rep->user->category_id)->category_name ?? '-' : '-';
    return view('slide',compact('rep','catName'));
  }

  public function my(){
    if(!session('user_id')) return redirect('/');
    $reports = Report::where('user_id',session('user_id'))->with('photos')->get();
    $cats = Category::all();
    return view('myreports',compact('reports','cats'));
  }

  public function store(Request $r){
    $r->validate(['title'=>'required','description'=>'required','date'=>'required','status'=>'required|in:0,1']);
    $u = User::find(session('user_id'));
    $rep = Report::create([
      'user_id'=>$u->id,'category_id'=>$u->category_id,
      'title'=>$r->title,'description'=>$r->description,'date'=>$r->date,'status'=>$r->status
    ]);
    return redirect('/manage/'.$rep->id);
  }

  public function destroy($id){
    $rep = Report::where('id',$id)->where('user_id',session('user_id'))->firstOrFail();
    foreach($rep->photos as $p){ if(file_exists(public_path($p->path))) unlink(public_path($p->path)); }
    $rep->delete();
    return redirect('/my-reports');
  }

  public function manage($id){
    if(!session('user_id')) return redirect('/');
    $rep = Report::with('photos')->where('id',$id)->where('user_id',session('user_id'))->firstOrFail();
    $catName = $rep->category_id ? Category::find($rep->category_id)->category_name ?? '-' : '-';
    return view('manage',compact('rep','catName'));
  }

  public function update(Request $r,$id){
    $r->validate(['title'=>'required','description'=>'required','date'=>'required','status'=>'required|in:0,1']);
    $rep = Report::where('id',$id)->where('user_id',session('user_id'))->firstOrFail();
    $rep->update($r->only(['title','description','date','status']));
    return redirect('/my-reports');
  }
}

<?php
namespace App\Http\Controllers;
use Illuminate\Http\Request;
use App\Models\Report;
use App\Models\Photo;

class PhotoController extends Controller {
  public function upload(Request $r,$id){
    $rep = Report::where('id',$id)->where('user_id',session('user_id'))->firstOrFail();
    $r->validate(['photos'=>'required','photos.*'=>'required|image|max:5120']);
    $out=[];
    if(!is_dir(public_path('uploads'))) mkdir(public_path('uploads'),0777,true);
    if(!$r->hasFile('photos')) return response()->json($out);
    foreach($r->file('photos') as $f){
      $name = 'uploads/'.time().'_'.uniqid().'.'.$f->getClientOriginalExtension();
      $full = public_path($name);
      $this->resize($f->getPathname(),$full,600,600);
      $ph = Photo::create(['report_id'=>$rep->id,'path'=>$name]);
      $out[]=['id'=>$ph->id,'path'=>$name];
    }
    return response()->json($out);
  }

  public function del($pid){
    $ph = Photo::findOrFail($pid);
    $rep = Report::where('id',$ph->report_id)->where('user_id',session('user_id'))->firstOrFail();
    if(file_exists(public_path($ph->path))) unlink(public_path($ph->path));
    $ph->delete();
    return response()->json(['ok'=>1]);
  }

  private function resize($src,$dst,$maxW,$maxH){
    [$w,$h,$type] = getimagesize($src);
    $ratio = min($maxW/$w,$maxH/$h,1);
    $nw=(int)($w*$ratio); $nh=(int)($h*$ratio);
    switch($type){
      case IMAGETYPE_JPEG: $s=imagecreatefromjpeg($src); break;
      case IMAGETYPE_PNG: $s=imagecreatefrompng($src); break;
      case IMAGETYPE_GIF: $s=imagecreatefromgif($src); break;
      case IMAGETYPE_WEBP: $s=imagecreatefromwebp($src); break;
      default: copy($src,$dst); return;
    }
    $d=imagecreatetruecolor($nw,$nh);
    if($type==IMAGETYPE_PNG||$type==IMAGETYPE_WEBP){ imagealphablending($d,false); imagesavealpha($d,true); }
    imagecopyresampled($d,$s,0,0,0,0,$nw,$nh,$w,$h);
    switch($type){
      case IMAGETYPE_JPEG: imagejpeg($d,$dst,85); break;
      case IMAGETYPE_PNG: imagepng($d,$dst); break;
      case IMAGETYPE_GIF: imagegif($d,$dst); break;
      case IMAGETYPE_WEBP: imagewebp($d,$dst,85); break;
    }
    imagedestroy($s); imagedestroy($d);
  }
}

// app/Http/Controllers/UserController.php
<?php
namespace App\Http\Controllers;
use App\Models\User;
use App\Models\Report;
use App\Models\Category;
use App\Models\Photo;

class UserController extends Controller {
  public function index(){
    if(session('role')!=1) return redirect('/');
    $users = User::where('role',0)->get();
    foreach($users as $u){
      $u->cat = $u->category_id ? Category::find($u->category_id)->category_name ?? '-' : '-';
      $u->rcount = Report::where('user_id',$u->id)->count();
      $rids = Report::where('user_id',$u->id)->pluck('id');
      $u->pcount = $rids->isEmpty() ? 0 : Photo::whereIn('report_id',$rids)->count();
    }
    return view('users',compact('users'));
  }

  public function destroy($id){
    if(session('role')!=1) return redirect('/');
    $u = User::findOrFail($id);
    foreach(Report::where('user_id',$u->id)->with('photos')->get() as $rep){
      foreach($rep->photos as $p){ if(file_exists(public_path($p->path))) unlink(public_path($p->path)); }
    }
    $u->delete();
    return back();
  }
}
